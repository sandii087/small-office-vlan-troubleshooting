! SW1 — Catalyst 2960; access ports for IT and Operations plus router trunk
! Fa0/1-2: IT VLAN 10; Fa0/3-4: Operations VLAN 20; Gi0/1: trunk to R1.
enable
configure terminal
hostname SW1
no ip domain-lookup

vlan 10
 name IT
exit
vlan 20
 name OPERATIONS
exit

interface range FastEthernet0/1 - 2
 description IT endpoint
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
exit

interface range FastEthernet0/3 - 4
 description Operations endpoint
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
exit

interface GigabitEthernet0/1
 description 802.1Q trunk to R1 Gi0/0
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
exit

end
write memory
